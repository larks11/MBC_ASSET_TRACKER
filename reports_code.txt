import Asset from "../models/Asset.js";
import Transaction from "../models/Transaction.js";
import RepairTicket from "../models/RepairTicket.js";
import asyncHandler from "../utils/asyncHandler.js";
import httpError from "../utils/httpError.js";
import { masterlistXlsx, repairLogXlsx, disposalLogXlsx } from "../exporters/excel.js";
import { memorandumReceiptDocx } from "../exporters/docx.js";

const XLSX = "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";
const DOCX = "application/vnd.openxmlformats-officedocument.wordprocessingml.document";

const send = (res, buffer, filename, type) => {
  res.setHeader("Content-Type", type);
  res.setHeader("Content-Disposition", `attachment; filename="${filename}"`);
  res.send(buffer);
};

const itemPopulate = { path: "items.asset", select: "assetId assetType brand model serialNo" };

// GET /exports/masterlist?assetType=&status=   â†’ MBC-SP-ICT-06F1
export const masterlist = asyncHandler(async (req, res) => {
  const { assetType, status } = req.query;
  const filter = { category: "Non-Consumable" };
  if (assetType) filter.assetType = assetType;
  if (status) filter.status = status;
  const assets = await Asset.find(filter).populate("currentUser", "name").sort("assetType assetId").lean();
  send(res, await masterlistXlsx(assets), "IT-Asset-Masterlist.xlsx", XLSX);
});

// GET /exports/memorandum-receipt/:id   â†’ MBC-SP-WHS-06F1 (.docx)
export const memorandumReceipt = asyncHandler(async (req, res) => {
  const txn = await Transaction.findById(req.params.id).populate(itemPopulate).lean();
  if (!txn) throw httpError(404, "Transaction not found");
  if (txn.type !== "Deploy") throw httpError(400, "A Memorandum Receipt can only be generated for a Deploy transaction");
  send(res, memorandumReceiptDocx(txn), `Memorandum-Receipt-${txn.transactionNo}.docx`, DOCX);
});

// GET /exports/repairs?status=&from=&to=   â†’ MBC-SP-ICT-06F3
export const repairs = asyncHandler(async (req, res) => {
  const { status, from, to } = req.query;
  const filter = {};
  if (status) filter.status = status;
  if (from || to) {
    filter.dateReceived = {};
    if (from) filter.dateReceived.$gte = new Date(from);
    if (to) filter.dateReceived.$lte = new Date(new Date(to).setHours(23, 59, 59, 999));
  }
  const tickets = await RepairTicket.find(filter).sort("dateReceived").lean();
  send(res, await repairLogXlsx(tickets), "IT-Asset-Repair-Monitoring.xlsx", XLSX);
});

// GET /exports/disposal/:id   â†’ MBC-SP-ICT-06F4 (usa ka Disposal transaction = usa ka task)
export const disposal = asyncHandler(async (req, res) => {
  const txn = await Transaction.findById(req.params.id).populate(itemPopulate).lean();
  if (!txn) throw httpError(404, "Transaction not found");
  if (txn.type !== "Disposal") throw httpError(400, "Not a disposal transaction");
  send(res, await disposalLogXlsx(txn), `IT-Asset-Disposal-Log-${txn.taskNo || txn.transactionNo}.xlsx`, XLSX);
});


===== backend\controllers\reportController.js =====

import Transaction from "../models/Transaction.js";
import asyncHandler from "../utils/asyncHandler.js";
import httpError from "../utils/httpError.js";

const REPORTS = { disposed: ["Disposal"], returned: ["Return", "TurnOver"] };

// GET /reports/:name  (disposed | returned) â†’ naka-flatten nga items kada transaction
export const flattened = asyncHandler(async (req, res) => {
  const types = REPORTS[req.params.name];
  if (!types) throw httpError(404, "Unknown report");

  const { page = 1, limit = 10, search = "", sort = "-date" } = req.query;

  const txns = await Transaction.find({ type: { $in: types }, status: "Completed" })
    .populate("items.asset", "assetId assetType brand model serialNo")
    .lean();

  let rows = txns.flatMap((t) =>
    t.items.map((it, i) => ({
      _id: `${t._id}-${i}`,
      transactionId: t._id,
      transactionNo: t.transactionNo,
      type: t.type,
      date: t.disposalDate || t.completedAt || t.date,
      taskNo: t.taskNo || "",
      ticketNo: it.ticketNo || t.ticketNo || "",
      employee: t.employeeSnapshot?.name || "",
      condition: it.condition,
      meansOfDisposal: it.meansOfDisposal || "",
      assetId: it.asset?.assetId || "",
      assetType: it.asset?.assetType || "",
      brand: it.asset?.brand || "",
      model: it.asset?.model || "",
      serialNo: it.asset?.serialNo || "",
    }))
  );

  const q = String(search).trim().toLowerCase();
  if (q) rows = rows.filter((r) => Object.values(r).join(" ").toLowerCase().includes(q));

  const field = String(sort).replace(/^-/, "");
  const dir = String(sort).startsWith("-") ? -1 : 1;
  rows.sort((a, b) => {
    const x = a[field] instanceof Date ? a[field].getTime() : (a[field] ?? "");
    const y = b[field] instanceof Date ? b[field].getTime() : (b[field] ?? "");
    return (x > y ? 1 : x < y ? -1 : 0) * dir;
  });

  const p = Math.max(parseInt(page, 10) || 1, 1);
  const l = Math.min(Math.max(parseInt(limit, 10) || 10, 1), 1000);
  res.json({
    data: rows.slice((p - 1) * l, p * l),
    meta: { total: rows.length, page: p, limit: l, pages: Math.ceil(rows.length / l) || 1 },
  });
});

import ExcelJS from "exceljs";
import path from "path";
import { fileURLToPath } from "url";
import httpError from "../utils/httpError.js";

const TPL = path.join(path.dirname(fileURLToPath(import.meta.url)), "..", "templates");

const longDate = (d) =>
  d ? new Date(d).toLocaleDateString("en-US", { year: "numeric", month: "long", day: "2-digit" }) : "";
const shortDate = (d) =>
  d ? new Date(d).toLocaleDateString("en-US", { year: "numeric", month: "2-digit", day: "2-digit" }) : "";

async function loadTemplate(file) {
  const wb = new ExcelJS.Workbook();
  await wb.xlsx.readFile(path.join(TPL, file));
  return { wb, ws: wb.worksheets[0] };
}

const cloneStyle = (cell) => JSON.parse(JSON.stringify(cell.style || {}));
const output = async (wb) => Buffer.from(await wb.xlsx.writeBuffer());

/* ------------------------------------------------------------------ *
 * MBC-SP-ICT-06F1  IT Asset Masterlist
 * Header row 6, data starts row 7. Ang template naay 28 ka rows (7-34);
 * kung mas daghan ang assets, gi-extend ang table nga parehas ang style.
 * ------------------------------------------------------------------ */
export async function masterlistXlsx(assets) {
  const { wb, ws } = await loadTemplate("MBC-SP-ICT-06F1.xlsx");
  const FIRST = 7, TEMPLATE_LAST = 34, COLS = 9;

  ws.getCell("C4").value = longDate(new Date()); // "Date as of:"

  const n = assets.length;
  const last = Math.max(FIRST + n - 1, TEMPLATE_LAST);

  if (last > TEMPLATE_LAST) {
    const innerStyles = Array.from({ length: COLS }, (_, i) => cloneStyle(ws.getRow(TEMPLATE_LAST - 1).getCell(i + 1)));
    const bottomStyles = Array.from({ length: COLS }, (_, i) => cloneStyle(ws.getRow(TEMPLATE_LAST).getCell(i + 1)));
    const height = ws.getRow(TEMPLATE_LAST - 1).height;
    for (let r = TEMPLATE_LAST; r <= last; r++) {
      const row = ws.getRow(r);
      if (height) row.height = height;
      for (let c = 1; c <= COLS; c++) row.getCell(c).style = r === last ? bottomStyles[c - 1] : innerStyles[c - 1];
    }
  }

  assets.forEach((a, i) => {
    const row = ws.getRow(FIRST + i);
    row.getCell(1).value = i + 1;
    row.getCell(2).value = a.assetId || "";
    row.getCell(3).value = a.assetType || "";
    row.getCell(4).value = a.brand || "";
    row.getCell(5).value = a.model || "";
    row.getCell(6).value = a.serialNo || "";
    row.getCell(7).value = a.currentUser?.name || "";
    row.getCell(8).value = a.location || "";
    row.getCell(9).value = a.status || "";
  });

  ws.pageSetup.printArea = `A1:I${last}`;
  return output(wb);
}

/* ------------------------------------------------------------------ *
 * MBC-SP-ICT-06F3  Repair & Service Request Monitoring
 * Data rows 6-24 (19 rows). D:E, F:J ug K:L naka-merge na sa template.
 * ------------------------------------------------------------------ */
export const REPAIR_FORM_CAPACITY = 19;

export async function repairLogXlsx(tickets) {
  if (tickets.length > REPAIR_FORM_CAPACITY) {
    throw httpError(
      400,
      `The repair monitoring form fits ${REPAIR_FORM_CAPACITY} tickets per page but ${tickets.length} matched. Narrow the date range or status filter.`
    );
  }
  const { wb, ws } = await loadTemplate("MBC-SP-ICT-06F3.xlsx");
  ws.getCell("C3").value = longDate(new Date());

  tickets.forEach((t, i) => {
    const row = ws.getRow(6 + i);
    row.getCell("A").value = i + 1;
    row.getCell("B").value = t.ticketNo || "";
    row.getCell("C").value = shortDate(t.dateReceived);
    row.getCell("D").value = t.assetCode || "";
    row.getCell("F").value = t.assetType || "";
    row.getCell("K").value = t.problem || "";
    row.getCell("M").value = t.solution || "";
    row.getCell("N").value = shortDate(t.dateStarted);
    row.getCell("O").value = shortDate(t.dateFinished);
    row.getCell("P").value = t.status || "";
  });
  return output(wb);
}

/* ------------------------------------------------------------------ *
 * MBC-SP-ICT-06F4  IT Asset Disposal Log  (usa ka Disposal transaction = usa ka form)
 * Data rows 7-35 (29 rows).
 * ------------------------------------------------------------------ */
export const DISPOSAL_FORM_CAPACITY = 29;

export async function disposalLogXlsx(txn) {
  const items = txn.items || [];
  if (items.length > DISPOSAL_FORM_CAPACITY) {
    throw httpError(400, `The disposal log fits ${DISPOSAL_FORM_CAPACITY} items per page but this task has ${items.length}.`);
  }
  const { wb, ws } = await loadTemplate("MBC-SP-ICT-06F4.xlsx");
  ws.getCell("C3").value = longDate(txn.disposalDate || txn.completedAt || txn.date);
  ws.getCell("C4").value = txn.taskNo || txn.transactionNo || "";

  items.forEach((it, i) => {
    const a = it.asset || {};
    const row = ws.getRow(7 + i);
    row.getCell("A").value = i + 1;
    row.getCell("B").value = it.ticketNo || "";
    row.getCell("C").value = a.assetId || "";
    row.getCell("D").value = a.assetType || "";
    row.getCell("E").value = a.brand || "";
    row.getCell("F").value = a.model || "";
    row.getCell("G").value = a.serialNo || "";
    row.getCell("H").value = it.meansOfDisposal || "";
  });
  return output(wb);
}


===== backend\exporters\docx.js =====

import fs from "fs";
import path from "path";
import { fileURLToPath } from "url";
import PizZip from "pizzip";
import Docxtemplater from "docxtemplater";

const TPL = path.join(path.dirname(fileURLToPath(import.meta.url)), "..", "templates");

const longDate = (d) =>
  d ? new Date(d).toLocaleDateString("en-US", { year: "numeric", month: "long", day: "2-digit" }) : "";

/**
 * MBC-SP-WHS-06F1 Memorandum Receipt.
 * Ang gamit nga template kay ang imong original nga naay {tags} na sa blank cells
 * (MBC-SP-WHS-06F1.tagged.docx), mao nga nagpabilin ang logo, header, terms ug signature blocks.
 * txn kinahanglan naay populated nga items.asset.
 */
export function memorandumReceiptDocx(txn) {
  const emp = txn.employeeSnapshot || {};
  const data = {
    transactionNo: txn.transactionNo || "",
    dateIssued: longDate(txn.date),
    employeeName: (emp.name || "").toUpperCase(),
    position: emp.position || "",
    department: [txn.department || emp.department, txn.location].filter(Boolean).filter((v, i, a) => a.indexOf(v) === i).join(" / "),
    items: (txn.items || []).map((it) => {
      const a = it.asset || {};
      return {
        assetType: (a.assetType || "").toUpperCase(),
        assetId: a.assetId || "",
        qty: it.qty ?? 1,
        brandModel: [a.brand, a.model].filter(Boolean).join(" / "),
        serialNo: a.serialNo || "",
        details: it.details || "",
      };
    }),
    issuedByName: (txn.issuedBy?.name || "").toUpperCase(),
    notedByName: (txn.notedBy?.name || "").toUpperCase(),
  };

  const zip = new PizZip(fs.readFileSync(path.join(TPL, "MBC-SP-WHS-06F1.tagged.docx")));
  const doc = new Docxtemplater(zip, { paragraphLoop: true, linebreaks: true, nullGetter: () => "" });
  doc.render(data);
  return doc.getZip().generate({ type: "nodebuffer", compression: "DEFLATE" });
}

import { useEffect, useMemo, useState } from "react";
import { Navigate, useParams } from "react-router-dom";
import { FileSpreadsheet, FileText, Printer } from "lucide-react";
import usePaginated from "../../hooks/usePaginated.js";
import DataTable from "../../components/ui/DataTable.jsx";
import FilterSelect from "../../components/ui/FilterSelect.jsx";
import Button from "../../components/ui/Button.jsx";
import Notice from "../../components/ui/Notice.jsx";
import { employeeApi } from "../../services/index.js";
import { downloadCSV, downloadFile, printTable } from "../../utils/exportClient.js";
import { REPORTS } from "./reportConfig.jsx";

const clean = (o) => Object.fromEntries(Object.entries(o).filter(([, v]) => v !== "" && v != null));

export default function ReportView() {
  const { reportId } = useParams();
  const cfg = REPORTS.find((r) => r.id === reportId);
  if (!cfg) return <Navigate to="/reports/masterlist" replace />;
  return <ReportBody key={cfg.id} cfg={cfg} />;
}

function ReportBody({ cfg }) {
  const table = usePaginated((p, s) => cfg.source(p, s), { initialSort: cfg.sort, initialFilters: cfg.base || {} });
  const [employees, setEmployees] = useState([]);
  const [busy, setBusy] = useState("");
  const [error, setError] = useState("");

  const needsEmployees = cfg.filters.some((f) => f.type === "employee");
  useEffect(() => {
    if (!needsEmployees) return;
    employeeApi.list({ limit: 1000, sort: "name" }).then((res) => setEmployees(res.data.map((e) => ({ value: e._id, label: e.name })))).catch(() => {});
  }, [needsEmployees]);

  const run = async (key, fn) => {
    setBusy(key);
    setError("");
    try {
      await fn();
    } catch (e) {
      setError(e.message);
    } finally {
      setBusy("");
    }
  };

  const ctx = { busy, download: (path, params, filename, key) => run(key || path, () => downloadFile(path, params, filename)) };
  const columns = useMemo(() => cfg.columns(ctx), [cfg, busy]); // eslint-disable-line react-hooks/exhaustive-deps

  const query = () => ({ ...clean(table.filters), search: table.search || undefined, sort: table.sort || undefined, page: 1, limit: 1000 });
  const fetchAll = async () => (await cfg.source(query())).data;

  const pick = (obj, keys) => Object.fromEntries(keys.map((k) => [k, obj[k]]).filter(([, v]) => v));

  const exportBar = (
    <>
      {cfg.excel && (
        <Button icon={FileSpreadsheet} disabled={busy === "excel"} onClick={() => run("excel", () => downloadFile(cfg.excel.path, pick(table.filters, cfg.excel.params), cfg.excel.filename))}>
          {busy === "excel" ? "Preparing..." : cfg.excel.label}
        </Button>
      )}
      <Button variant="secondary" icon={FileText} disabled={busy === "csv"} onClick={() => run("csv", async () => downloadCSV(`${cfg.id}.csv`, columns, await fetchAll()))}>
        CSV
      </Button>
      <Button variant="secondary" icon={Printer} disabled={busy === "print"} onClick={() => run("print", async () => printTable(cfg.title, columns, await fetchAll()))}>
        Print
      </Button>
    </>
  );

  return (
    <section>
      <div className="mb-4">
        <h2 className="text-lg font-semibold text-navy-900">{cfg.title}</h2>
        <p className="mt-0.5 max-w-prose text-sm text-slate-500">{cfg.description}</p>
      </div>
      <Notice tone="error" onClose={() => setError("")}>{error}</Notice>

      <DataTable
        {...table}
        columns={columns}
        searchPlaceholder={cfg.search}
        toolbarRight={exportBar}
        filters={
          <>
            {cfg.filters.map((f) => (
              <FilterSelect
                key={f.key}
                value={table.filters[f.key]}
                onChange={(v) => table.setFilter(f.key, v)}
                options={f.type === "employee" ? employees : f.options}
                placeholder={f.placeholder}
              />
            ))}
            {cfg.dates && (
              <>
                <DateInput label="From" value={table.filters.from} onChange={(v) => table.setFilter("from", v)} />
                <DateInput label="To" value={table.filters.to} onChange={(v) => table.setFilter("to", v)} />
              </>
            )}
          </>
        }
      />
    </section>
  );
}

function DateInput({ label, value, onChange }) {
  return (
    <label className="flex items-center gap-1.5 text-sm text-slate-500">
      {label}
      <input
        type="date"
        value={value || ""}
        onChange={(e) => onChange(e.target.value)}
        className="rounded border border-line bg-white px-2 py-1.5 text-sm text-slate-700 focus:border-brand-500 focus:outline-none"
      />
    </label>
  );
}


===== frontend\src\utils\exportClient.js =====

import api from "../services/api.js";

const esc = (v) => `"${String(v ?? "").replace(/"/g, '""')}"`;
const html = (v) => String(v ?? "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

export const cellText = (col, row) => (col.text ? col.text(row) : row[col.key]) ?? "";
const dataColumns = (columns) => columns.filter((c) => c.key !== "actions");

function triggerDownload(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

// File gikan sa backend (.xlsx / .docx), gamit ang imong original nga templates
export async function downloadFile(path, params, filename) {
  const blob = await api.get(path, { params, responseType: "blob" });
  triggerDownload(blob, filename);
}

export function downloadCSV(filename, columns, rows) {
  const cols = dataColumns(columns);
  const lines = [
    cols.map((c) => esc(c.header)).join(","),
    ...rows.map((r) => cols.map((c) => esc(cellText(c, r))).join(",")),
  ];
  triggerDownload(new Blob(["\ufeff" + lines.join("\r\n")], { type: "text/csv;charset=utf-8" }), filename);
}

export function printTable(title, columns, rows) {
  const cols = dataColumns(columns);
  const w = window.open("", "_blank");
  if (!w) throw new Error("The browser blocked the print window. Allow pop-ups for this site and try again.");
  w.document.write(`<!doctype html><html><head><title>${html(title)}</title><style>
    body{font-family:Arial,sans-serif;color:#111;margin:24px}
    h1{font-size:18px;margin:0}.sub{color:#555;font-size:12px;margin:2px 0 14px}
    table{border-collapse:collapse;width:100%;font-size:11px}
    th,td{border:1px solid #999;padding:4px 6px;text-align:left}th{background:#eef2f9}
    @media print{@page{size:landscape;margin:12mm}}
  </style></head><body>
    <h1>MAC Builders Corp.</h1>
    <div class="sub">ICT Asset Monitoring System Â· ${html(title)} Â· ${new Date().toLocaleString("en-PH")}</div>
    <table><thead><tr>${cols.map((c) => `<th>${html(c.header)}</th>`).join("")}</tr></thead>
    <tbody>${rows.map((r) => `<tr>${cols.map((c) => `<td>${html(cellText(c, r))}</td>`).join("")}</tr>`).join("")}</tbody></table>
  </body></html>`);
  w.document.close();
  w.focus();
  w.print();
}
